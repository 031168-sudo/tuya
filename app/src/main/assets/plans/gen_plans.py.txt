def wall_on(f, vertical, c, a, b):
    """Толщина стены, лежащей на линии (x=c или y=c) и перекрывающей отрезок [a,b]."""
    best = 0
    for x1, y1, x2, y2, t in f["walls"]:
        if vertical and x1 == x2 == c and min(y1, y2) < b and max(y1, y2) > a: best = max(best, t)
        if not vertical and y1 == y2 == c and min(x1, x2) < b and max(x1, x2) > a: best = max(best, t)
    return best

        w = (x2 - x1) - wall_on(f, True, x1, y1, y2) / 2 - wall_on(f, True, x2, y1, y2) / 2
        h = (y2 - y1) - wall_on(f, False, y1, x1, x2) / 2 - wall_on(f, False, y2, x1, x2) / 2
        r["area"] = round(w * h / 1e6, 1)

PALETTE = ["#F7C59F", "#A8DCC3", "#ABC4F0", "#F2B8C6", "#C9E4A6", "#C7B8EE", "#F9E2A2", "#A4D8E6"]

def svg(f, scale=0.07, temps=None):
    m = 60
    W = int(f["bounds"][2] * scale + 2 * m); H = int(f["bounds"][3] * scale + 2 * m + 50)
    X = lambda v: m + v * scale
    Y = lambda v: m + 30 + v * scale
    o = [f'<svg xmlns="http://www.w3.org/2000/svg" width="{W}" height="{H}" viewBox="0 0 {W} {H}" font-family="Inter, Roboto, DejaVu Sans, sans-serif">',
         f'<rect width="{W}" height="{H}" fill="#FBFAF7"/>',
         f'<text x="{m}" y="{m - 8}" font-size="22" font-weight="600" fill="#2B2B2B">{f["title"]}</text>']
    # крыльцо
    for t in f["terraces"]:
        x1, y1, x2, y2 = t["rect"]
        o.append(f'<rect x="{X(x1)}" y="{Y(y1)}" width="{(x2-x1)*scale}" height="{(y2-y1)*scale}" fill="#EFEBE3" stroke="#B9B1A3" stroke-dasharray="5 4" stroke-width="1.5"/>')
        o.append(f'<text x="{X((x1+x2)/2)}" y="{Y((y1+y2)/2)+5}" text-anchor="middle" font-size="13" fill="#8A8273">{t["name"]}</text>')
    # комнаты
    for i, r in enumerate(f["rooms"]):
        x1, y1, x2, y2 = r["rect"]
        fill = PALETTE[i % len(PALETTE)] if temps is None else temps.get(r["id"], "#E5E5E5")
        o.append(f'<rect x="{X(x1)}" y="{Y(y1)}" width="{(x2-x1)*scale}" height="{(y2-y1)*scale}" fill="{fill}"/>')
    # стены
    for x1, y1, x2, y2, t in f["walls"]:
        h = t / 2
        if y1 == y2:
            o.append(f'<rect x="{X(min(x1,x2)-h)}" y="{Y(y1-h)}" width="{(abs(x2-x1)+t)*scale}" height="{t*scale}" fill="#33363B"/>')
        else:
            o.append(f'<rect x="{X(x1-h)}" y="{Y(min(y1,y2)-h)}" width="{t*scale}" height="{(abs(y2-y1)+t)*scale}" fill="#33363B"/>')
    # окна
    for x1, y1, x2, y2 in f["windows"]:
        # проём во всю толщину стены, стёкла — на 1/10 толщины от оси
        T = wall_on(f, x1 == x2, x1 if x1 == x2 else y1, min(y1, y2) if x1 == x2 else min(x1, x2), max(y1, y2) if x1 == x2 else max(x1, x2)) or OUT
        g = T / 10
        if y1 == y2:
            o.append(f'<rect x="{X(x1)}" y="{Y(y1-T/2)}" width="{(x2-x1)*scale}" height="{T*scale}" fill="#FFFFFF"/>')
            for d in (-g, g):
                o.append(f'<line x1="{X(x1)}" y1="{Y(y1+d)}" x2="{X(x2)}" y2="{Y(y1+d)}" stroke="#6FA8DC" stroke-width="1.6"/>')
        else:
            o.append(f'<rect x="{X(x1-T/2)}" y="{Y(y1)}" width="{T*scale}" height="{(y2-y1)*scale}" fill="#FFFFFF"/>')
            for d in (-g, g):
                o.append(f'<line x1="{X(x1+d)}" y1="{Y(y1)}" x2="{X(x1+d)}" y2="{Y(y2)}" stroke="#6FA8DC" stroke-width="1.6"/>')
    # двери: проём + полотно + дуга
    for x1, y1, x2, y2, side in f["doors"]:
        L = math.hypot(x2 - x1, y2 - y1)
        T = wall_on(f, x1 == x2, x1 if x1 == x2 else y1, min(y1, y2) if x1 == x2 else min(x1, x2), max(y1, y2) if x1 == x2 else max(x1, x2)) or OUT
        if y1 == y2:
            o.append(f'<rect x="{X(min(x1,x2))}" y="{Y(y1-T/2)}" width="{abs(x2-x1)*scale}" height="{T*scale}" fill="#FBFAF7"/>')
            ex, ey = x1, y1 + side * L
        else:
            o.append(f'<rect x="{X(x1-T/2)}" y="{Y(min(y1,y2))}" width="{T*scale}" height="{abs(y2-y1)*scale}" fill="#FBFAF7"/>')
            ex, ey = x1 + side * L, y1
        o.append(f'<line x1="{X(x1)}" y1="{Y(y1)}" x2="{X(ex)}" y2="{Y(ey)}" stroke="#33363B" stroke-width="2"/>')
        sweep = 1 if ((x2 - x1) * (ey - y1) - (y2 - y1) * (ex - x1)) < 0 else 0
        o.append(f'<path d="M {X(ex)} {Y(ey)} A {L*scale} {L*scale} 0 0 {sweep} {X(x2)} {Y(y2)}" fill="none" stroke="#9AA0A6" stroke-width="1" stroke-dasharray="3 3"/>')
    # подписи
    for r in f["rooms"]:
        x1, y1, x2, y2 = r["rect"]
        cx, cy = X((x1 + x2) / 2), Y((y1 + y2) / 2)
        fs = max(10, min(15, int(((x2 - x1) - 300) * scale / 7.5)))
        label = r["name"]
        if temps is not None and r["id"] in TEMP_VAL:
            o.append(f'<text x="{cx}" y="{cy - 4}" text-anchor="middle" font-size="{fs+8}" font-weight="700" fill="#2B2B2B">{str(TEMP_VAL[r["id"]]).replace(".", ",")}°</text>')
            o.append(f'<text x="{cx}" y="{cy + 16}" text-anchor="middle" font-size="{max(11, fs-1)}" fill="#3C3C3C">{label}</text>')
            o.append(f'<text x="{cx}" y="{cy + 31}" text-anchor="middle" font-size="{max(10, fs-3)}" fill="#5A5A5A">{str(r["area"]).replace(".", ",")} м²</text>')
        else:
            o.append(f'<text x="{cx}" y="{cy}" text-anchor="middle" font-size="{fs}" font-weight="600" fill="#2B2B2B">{label}</text>')
            o.append(f'<text x="{cx}" y="{cy + 17}" text-anchor="middle" font-size="{fs-2}" fill="#5A5A5A">{r["area"]:.1f} м²</text>'.replace(".", ",", 1) if False else
                     f'<text x="{cx}" y="{cy + 17}" text-anchor="middle" font-size="{fs-2}" fill="#5A5A5A">{str(r["area"]).replace(".", ",")} м²</text>')
    # размеры
    o.append(f'<line x1="{X(0)}" y1="{Y(f["bounds"][3])+44}" x2="{X(f["bounds"][2])}" y2="{Y(f["bounds"][3])+44}" stroke="#B0B0B0"/>')
    o.append(f'<text x="{X(f["bounds"][2]/2)}" y="{Y(f["bounds"][3])+38}" text-anchor="middle" font-size="11" fill="#8A8A8A">{f.get("dim") or format(f["bounds"][2], ",").replace(",", " ")}</text>')
    o.append('</svg>')
    return "\n".join(o)

# ---------- Цветовая шкала температур ----------
# Пастельные цвета, чтобы тёмный текст читался на любом: холодно — синий, комфорт — зелёный, жарко — красный.
T_STOPS = [(14, (120, 160, 235)), (18, (140, 205, 225)), (21, (155, 215, 160)), (23.5, (250, 205, 110)), (26, (238, 120, 95))]
T_MIN, T_MAX = T_STOPS[0][0], T_STOPS[-1][0]

def tcolor(t):
    t = max(T_MIN, min(T_MAX, t))
    for (a, ca), (b, cb) in zip(T_STOPS, T_STOPS[1:]):
        if a <= t <= b:
            k = (t - a) / (b - a)
            return "#%02x%02x%02x" % tuple(round(ca[i] + (cb[i] - ca[i]) * k) for i in range(3))

def legend_svg(h=420):
    """Вертикальная шкала: сверху жарко, снизу холодно."""
    W, top, bar_x, bar_w = 200, 70, 30, 26
    H = top + h + 40
    o = [f'<svg xmlns="http://www.w3.org/2000/svg" width="{W}" height="{H}" viewBox="0 0 {W} {H}" font-family="Inter, Roboto, DejaVu Sans, sans-serif">',
         f'<rect width="{W}" height="{H}" fill="#FBFAF7"/>',
         f'<text x="{bar_x - 4}" y="{top - 30}" font-size="15" font-weight="600" fill="#2B2B2B">Температура</text>',
         '<defs><linearGradient id="tg" x1="0" y1="1" x2="0" y2="0">']
    for t, c in T_STOPS:
        o.append(f'<stop offset="{(t - T_MIN) / (T_MAX - T_MIN):.3f}" stop-color="#%02x%02x%02x"/>' % c)
    o.append('</linearGradient></defs>')
    o.append(f'<rect x="{bar_x}" y="{top}" width="{bar_w}" height="{h}" rx="6" fill="url(#tg)"/>')
    ty = lambda t: top + h - (t - T_MIN) / (T_MAX - T_MIN) * h
    for t in range(T_MIN, T_MAX + 1, 2):
        lab = f"≤{t}°" if t == T_MIN else f"≥{t}°" if t == T_MAX else f"{t}°"
        o.append(f'<line x1="{bar_x + bar_w}" y1="{ty(t)}" x2="{bar_x + bar_w + 6}" y2="{ty(t)}" stroke="#6A6A6A"/>')
        o.append(f'<text x="{bar_x + bar_w + 10}" y="{ty(t) + 4}" font-size="12" fill="#3C3C3C">{lab}</text>')
    for t, name in ((25, "жарко"), (21, "комфорт"), (16, "холодно")):
        o.append(f'<text x="{bar_x + bar_w + 52}" y="{ty(t) + 4}" font-size="12" fill="#8A8A8A">{name}</text>')
    o.append('</svg>')
    return "\n".join(o)
